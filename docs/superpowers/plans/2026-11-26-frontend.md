**Fact sheet (the only facts the copy may state):** Hidden Hand is a 90-second on-chain trading pit on Solana devnet. Four seats; each gets two sealed cards (1–15) dealt by MagicBlock VRF inside a Private Ephemeral Rollup (a TEE), readable only by their owner. Everyone trades one contract worth the sum of all eight cards against an on-chain market maker; trades cost zero fees on the Ephemeral Rollup and need no wallet prompt. One contract moves the price one point, so each trade's size encodes what a sealed hand knows: this is a **linear information market**, and the price converges on the hidden sum by design, not by clever bots. At the bell the hands flip on-chain, the hidden sum draws across the chart, and the best trader takes the pot (0.01 devnet SOL per seat). Bots fill empty seats. Devnet only, no real funds.

`web/src/shared/pitTimeline.ts`:
```ts
import { decide, mulberry32, personaFor, type BotMemory, type BotPersona } from "@hidden-hand/bots/policy";
import { CARDS_PER_SEAT, SEATS, START_PRICE_C, hiddenSum, quote, type Side } from "@hidden-hand/client/engine";

export interface PitEvent { tMs: number; seat: number; side: Side; qty: number; priceAfterC: number }
export interface PitTimeline { cards: number[]; hiddenSum: number; events: PitEvent[]; closeC: number }

/** One illustrative round: the same policy and engine as the live bots, simulated with a seed (no network). */
export function pitTimeline(seed: number, roundMs: number): PitTimeline {
  const rng = mulberry32(seed);
  const cards: number[] = [];
  for (let i = 0; i < SEATS * CARDS_PER_SEAT; i++) cards.push(1 + Math.floor(rng() * 15));
  const pos = new Array<number>(SEATS).fill(0);
  const personas: BotPersona[] = [];
  const mems: BotMemory[] = [];
  const next: number[] = [];
  for (let s = 0; s < SEATS; s++) {
    personas.push(personaFor(s, rng));
    mems.push({ wobble: 0 });
    next.push(Math.floor(rng() * personas[s].intervalMaxMs));
  }
  let priceC = START_PRICE_C;
  const events: PitEvent[] = [];
  for (;;) {
    let s = 0;
    for (let k = 1; k < SEATS; k++) if (next[k] < next[s]) s = k;
    if (next[s] >= roundMs) break;
    const t = next[s];
    const own = cards[2 * s] + cards[2 * s + 1];
    const order = decide(own, pos[s], t, personas[s], mems[s], rng);
    if (order) {
      const f = quote(priceC, pos[s], order.side, order.qty);
      if (typeof f !== "string") {
        priceC = f.priceAfterC;
        pos[s] += f.posDelta;
        events.push({ tMs: t, seat: s, side: order.side, qty: order.qty, priceAfterC: priceC });
      }
    }
    const p = personas[s];
    next[s] = t + Math.floor(p.intervalMinMs + rng() * (p.intervalMaxMs - p.intervalMinMs));
  }
  return { cards, hiddenSum: hiddenSum(cards), events, closeC: priceC };
}
```

`web/src/shared/PriceChart.tsx` — an SVG chart drawn only with the dealt tokens (`var(--accent)` for the price line, `var(--ink)` at reduced opacity for the axis and labels, the dashed hidden-sum line in `var(--ink)`); y range 8–120 points with gridlines at 32/64/96; the x axis is time over `durationMs`; when `revealSum` is a number, a dashed horizontal line at that value draws in from left to right (SVG `stroke-dasharray` + `stroke-dashoffset` transition, at rest under `prefers-reduced-motion`) with the label "hidden sum <n>" in Chivo Mono:
```tsx
export interface ChartPoint { tMs: number; priceC: number }
export interface PriceChartProps { points: ChartPoint[]; durationMs: number; revealSum: number | null; height?: number; label: string }
const W = 640;
const Y_MIN = 8;
const Y_MAX = 120;

export function PriceChart({ points, durationMs, revealSum, height = 260, label }: PriceChartProps) {
  const x = (t: number) => (Math.min(t, durationMs) / durationMs) * W;
  const y = (p: number) => height - ((p - Y_MIN) / (Y_MAX - Y_MIN)) * height;
  const path = [{ tMs: 0, priceC: 6400 }, ...points]
    .map((pt, i) => `${i === 0 ? "M" : "L"}${x(pt.tMs).toFixed(1)},${y(pt.priceC / 100).toFixed(1)}`)
    .join(" ");
  const last = points.length > 0 ? points[points.length - 1].priceC / 100 : 64;
  return (
    <figure className="pit-chart" aria-label={label}>
      <svg viewBox={`0 0 ${W} ${height}`} role="img" aria-label={`${label}: price ${last.toFixed(2)}`}>
        {[32, 64, 96].map((g) => (
          <line key={g} x1={0} x2={W} y1={y(g)} y2={y(g)} className="pit-chart__grid" />
        ))}
        <path d={path} className="pit-chart__price" />
        {revealSum !== null && (
          <g className="pit-chart__reveal">
            <line x1={0} x2={W} y1={y(revealSum)} y2={y(revealSum)} className="pit-chart__sum" />
            <text x={W - 4} y={y(revealSum) - 6} textAnchor="end" className="pit-chart__sum-label">{`hidden sum ${revealSum}`}</text>
          </g>
        )}
      </svg>
    </figure>
  );
}
```

```ts
export interface UiViewLike { phase: "open" | "dealing" | "trading" | "revealed" | "settled"; seatCount: number; endTs: number }
export interface UiInput { roundParam: string | null; loading: boolean; error: string | null; view: UiViewLike | null; nowMs: number }
export type UiState =
  | { kind: "no-round" }
  | { kind: "loading" }
  | { kind: "error"; message: string }
  | { kind: "waiting-for-seats" }
  | { kind: "waiting-for-deal" }
  | { kind: "trading"; secondsLeft: number }
  | { kind: "bell" }
  | { kind: "revealed" }
  | { kind: "settled" };

export function deriveUiState(i: UiInput): UiState {
  if (i.roundParam === null) return { kind: "no-round" };
  if (i.error !== null) return { kind: "error", message: i.error };
  if (i.loading || i.view === null) return { kind: "loading" };
  const v = i.view;
  switch (v.phase) {
    case "open":
      return v.seatCount < 4 ? { kind: "waiting-for-seats" } : { kind: "waiting-for-deal" };
    case "dealing":
      return { kind: "waiting-for-deal" };
    case "trading": {
      const left = Math.ceil(v.endTs - i.nowMs / 1000);
      return left > 0 ? { kind: "trading", secondsLeft: left } : { kind: "bell" };
    }
    case "revealed":
      return { kind: "revealed" };
    case "settled":
      return { kind: "settled" };
  }
}
```

`web/src/app/burner.ts` (a devnet-only seat key kept in this browser; generated at runtime by the app, never by the build):
```ts
import { Keypair } from "@solana/web3.js";

const KEY = "hidden-hand-seat-key-v1";

export function loadSeatKey(): Keypair {
  try {
    const raw = window.localStorage.getItem(KEY);
    if (raw) return Keypair.fromSecretKey(Uint8Array.from(JSON.parse(raw) as number[]));
  } catch {
    // storage blocked or corrupt: fall through to a session-only key
  }
  const kp = Keypair.generate();
  try {
    window.localStorage.setItem(KEY, JSON.stringify(Array.from(kp.secretKey)));
  } catch {
    // session-only key
  }
  return kp;
}
```
