# Clients Implementation Plan

**Goal:** TypeScript clients for Hidden Hand: a shared client package (connections, PDAs, TEE auth, DealAdapter, round API, TS engine port, the Day-1 spike script) and a bots package (seeded policy with its Monte Carlo gate, the demo-round runner that writes the devnet proof).
**Architecture:** A pnpm workspace at the repo root with `client/` (`@hidden-hand/client`, browser-safe `.` entry plus a Node-only `./node` entry) and `bots/` (`@hidden-hand/bots`). TS sources are consumed directly (no build step); Node scripts run with `tsx`. Every devnet-touching script is run only by a human gate; executors only write and test it.
**Tech Stack:** Node 22 (Windows), pnpm 10.33.0, TypeScript ~5.8.3, vitest ^3.2.4, tsx ^4.20.3, `@coral-xyz/anchor` 0.32.1, `@solana/web3.js` ^1.99.0, `@magicblock-labs/ephemeral-rollups-sdk` 0.18.0, tweetnacl ^1.0.3, `@noble/hashes` ^1.8.0, `@types/node` ^22.15.0.

Context (the on-chain surface this task calls; Anchor TS names are camelCase):
- `openTable(roundId: BN, mode: {tee:{}}|{commitReveal:{}}, durationSecs: number, entryLamports: BN)` accounts `{ table, authority }` (base).
- `joinTable(commitment: number[32])` accounts `{ table, hand, vault, player }` (base; `player` signs and pays the entry + hand rent).
- `delegatePda(target: {table:{roundId}} | {hand:{table, owner}})` accounts `{ payer, validator, pda }` (base; the rest resolve from the IDL).
- `createTablePermission()` accounts `{ table, permission, payer, permissionProgram, ephemeralVault, magicProgram }`; `createHandPermission(owner)` accounts `{ table, hand, permission, payer, permissionProgram, ephemeralVault, magicProgram }` (TEE ER).
- `requestDeal(clientSeed: number)` accounts `{ payer, table, oracleQueue }` (ER). The `#[vrf]`-injected accounts resolve from the IDL; if `tsc` reports one missing, read its exact name in `client/src/idl/hidden_hand.json` and pass it: program identity = `PublicKey.findProgramAddressSync([Buffer.from("identity")], PROGRAM_ID)[0]`, VRF program `Vrf1RNUjXmQGjmQrQLvJHs9SNkvDJEsRVFPkfSQUwGz`, `SYSVAR_SLOT_HASHES_PUBKEY`, `SystemProgram.programId`.
- `revealHands()` accounts `{ table, hand0..hand3, permission0..permission3, payer, permissionProgram, ephemeralVault, magicProgram }` (TEE ER).
- Decoded `Table` fields (Anchor camelCase): `roundId: BN, authority, mode: {tee:{}}|{commitReveal:{}}, phase: {open:{}}|{dealing:{}}|{trading:{}}|{revealed:{}}|{settled:{}}, durationSecs, entryLamports: BN, seatCount, seats: {owner, hand, cashC: BN, pos, commitment: number[], revealed}[4], priceC, startTs: BN, endTs: BN, seed: number[], cards: number[], hiddenSum, winner (255 = none), pnlC: BN[], paid, tradeCount, trades: {seat, side (0 buy, 1 sell), qty, priceAfterC, ts: BN}[128]` (a ring: entry k lives at `k % 128`).
- TEE reads need an auth token per identity: `getAuthToken(TEE_RPC, pubkey, signMessage)` → `new Connection(TEE_RPC + "?token=" + token)`; wait with `waitUntilPermissionActive(TEE_RPC, account, timeoutMs)` after creating a permission (both from `@magicblock-labs/ephemeral-rollups-sdk` 0.18.0, along with `permissionPdaFromAccount`, `PERMISSION_PROGRAM_ID`, `MAGIC_PROGRAM_ID`).
- Keys: the burner is `HH_BURNER_KEYPAIR_JSON` (JSON array of 64 numbers) from the user env var, else the git-ignored repo-root `.env`. Never print a secret key. Seat keys are derived from the burner; nothing is generated or stored on disk.

```ts
import { sha256 } from "@noble/hashes/sha256";
import { Keypair } from "@solana/web3.js";

/** Seat keys derived from the burner: deterministic, never generated or stored, never printed. */
export function deriveBotKeypair(burner: Keypair, index: number): Keypair {
  const tag = new TextEncoder().encode(`hidden-hand-bot-${index}`);
  const buf = new Uint8Array(32 + tag.length);
  buf.set(burner.secretKey.slice(0, 32), 0);
  buf.set(tag, 32);
  return Keypair.fromSeed(sha256(buf));
}

export function keypairFromJson(json: string): Keypair {
  const parsed: unknown = JSON.parse(json);
  if (!Array.isArray(parsed) || parsed.length !== 64 || !parsed.every((n) => Number.isInteger(n) && n >= 0 && n <= 255)) {
    throw new Error("HH_BURNER_KEYPAIR_JSON must be a JSON array of 64 numbers");
  }
  return Keypair.fromSecretKey(Uint8Array.from(parsed as number[]));
}

export function parseDotEnv(text: string, name: string): string | null {
  for (const line of text.split(/\r?\n/)) {
    const i = line.indexOf("=");
    if (i > 0 && line.slice(0, i).trim() === name) {
      const value = line.slice(i + 1).trim();
      return value.length > 0 ? value : null;
    }
  }
  return null;
}
```

```ts
import { Connection, type Keypair } from "@solana/web3.js";
import nacl from "tweetnacl";
import { BASE_RPC, TEE_RPC } from "./config";

export function baseConnection(): Connection {
  return new Connection(BASE_RPC, "confirmed");
}

export function publicErConnection(rpc: string): Connection {
  return new Connection(rpc, { commitment: "confirmed", wsEndpoint: rpc.replace(/^http/, "ws") });
}

/** A TEE connection that reads as `identity` (its auth token decides which private accounts it sees). */
export async function teeConnection(identity: Keypair): Promise<{ connection: Connection; expiresAt: number }> {
  const auth = await getAuthToken(TEE_RPC, identity.publicKey, (message: Uint8Array) =>
    Promise.resolve(nacl.sign.detached(message, identity.secretKey)),
  );
  const url = `${TEE_RPC}?token=${auth.token}`;
  const ws = `${TEE_RPC.replace(/^http/, "ws")}?token=${auth.token}`;
  return { connection: new Connection(url, { commitment: "confirmed", wsEndpoint: ws }), expiresAt: auth.expiresAt };
}
```

```ts
import { AnchorProvider, Program } from "@coral-xyz/anchor";
import { type Connection, type Keypair, Transaction, VersionedTransaction } from "@solana/web3.js";
import type { HiddenHand } from "./idl/hidden_hand";
import idl from "./idl/hidden_hand.json";

type ProviderWallet = ConstructorParameters<typeof AnchorProvider>[1];

export function keypairWallet(kp: Keypair): ProviderWallet {
  const sign = <T extends Transaction | VersionedTransaction>(tx: T): T => {
    if (tx instanceof VersionedTransaction) tx.sign([kp]);
    else tx.partialSign(kp);
    return tx;
  };
  return {
    publicKey: kp.publicKey,
    payer: kp,
    signTransaction: async <T extends Transaction | VersionedTransaction>(tx: T) => sign(tx),
    signAllTransactions: async <T extends Transaction | VersionedTransaction>(txs: T[]) => txs.map(sign),
  } as ProviderWallet;
}

export function makeProgram(connection: Connection, kp: Keypair): Program<HiddenHand> {
  const provider = new AnchorProvider(connection, keypairWallet(kp), { commitment: "confirmed" });
  return new Program<HiddenHand>(idl as unknown as HiddenHand, provider);
}
```

```ts
  startTs: number; endTs: number; cards: number[]; hiddenSum: number; winner: number | null; pnlC: number[];
  paid: boolean; tradeCount: number; trades: TradeView[];
}

type RawTable = IdlAccounts<HiddenHand>["table"];
const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));

export function decodeTable(address: PublicKey, raw: RawTable): TableView {
  const ring = raw.trades.length;
  const count = raw.tradeCount;
  const n = Math.min(count, ring);
  const trades: TradeView[] = [];
  for (let k = count - n; k < count; k++) {
    const t = raw.trades[k % ring];
    trades.push({ index: k, seat: t.seat, side: t.side === 0 ? "buy" : "sell", qty: t.qty, priceAfterC: t.priceAfterC, ts: t.ts.toNumber() });
  }
  return {
    address,
    roundId: raw.roundId,
    authority: raw.authority,
    mode: Object.keys(raw.mode)[0] === "tee" ? "tee" : "commit-reveal",
    phase: Object.keys(raw.phase)[0] as PhaseName,
    durationSecs: raw.durationSecs,
    entryLamports: raw.entryLamports.toNumber(),
    seatCount: raw.seatCount,
    seats: raw.seats.map((s) => ({ owner: s.owner, hand: s.hand, cashC: s.cashC.toNumber(), pos: s.pos, revealed: s.revealed })),
    priceC: raw.priceC,
    startTs: raw.startTs.toNumber(),
    endTs: raw.endTs.toNumber(),
    cards: [...raw.cards],
    hiddenSum: raw.hiddenSum,
    winner: raw.winner === 255 ? null : raw.winner,
    pnlC: raw.pnlC.map((x) => x.toNumber()),
    paid: raw.paid,
    tradeCount: count,
    trades,
  };
}

export async function fetchTable(program: Program<HiddenHand>, table: PublicKey): Promise<TableView | null> {
  const raw = await program.account.table.fetchNullable(table);
  return raw ? decodeTable(table, raw) : null;
}

export async function waitForPhase(program: Program<HiddenHand>, table: PublicKey, phase: PhaseName, timeoutMs: number): Promise<TableView> {
  const deadline = Date.now() + timeoutMs;
  let last: TableView | null = null;
  while (Date.now() < deadline) {
    last = await fetchTable(program, table).catch(() => null);
    if (last && last.phase === phase) return last;
    await sleep(500);
  }
  throw new Error(`table ${table.toBase58()} did not reach phase ${phase} in ${timeoutMs} ms (last: ${last ? last.phase : "unreadable"})`);
}

export async function waitUntilOnEr(er: Connection, account: PublicKey, timeoutMs: number): Promise<void> {
  const deadline = Date.now() + timeoutMs;
  while (Date.now() < deadline) {
    const info = await er.getAccountInfo(account).catch(() => null);
    if (info) return;
    await sleep(500);
  }
  throw new Error(`${account.toBase58()} not visible on the ER after ${timeoutMs} ms`);
}

export async function waitForOwner(base: Connection, account: PublicKey, owner: PublicKey, timeoutMs: number): Promise<void> {
  const deadline = Date.now() + timeoutMs;
  while (Date.now() < deadline) {
    const info = await base.getAccountInfo(account, "confirmed").catch(() => null);
    if (info && info.owner.equals(owner)) return;
    await sleep(1000);
  }
  throw new Error(`${account.toBase58()} is not owned by ${owner.toBase58()} after ${timeoutMs} ms`);
}

/** Top up each target to `lamportsEach` from the payer in one base-layer tx; null when nobody needed funds. */
export async function fundAccounts(base: Connection, payer: Keypair, targets: PublicKey[], lamportsEach: number): Promise<string | null> {
  const ixs = [];
  for (const to of targets) {
    const have = await base.getBalance(to, "confirmed");
    if (have < lamportsEach) ixs.push(SystemProgram.transfer({ fromPubkey: payer.publicKey, toPubkey: to, lamports: lamportsEach - have }));
  }
  return ixs.length > 0 ? sendIxs(base, ixs, payer) : null;
}

export async function openTable(
  base: Program<HiddenHand>, authority: Keypair, roundId: BN, mode: DealMode, durationSecs: number, entryLamports: number,
): Promise<string> {
  const ix = await base.methods
    .openTable(roundId, mode === "tee" ? { tee: {} } : { commitReveal: {} }, durationSecs, new BN(entryLamports))
    .accountsPartial({ table: tablePda(roundId), authority: authority.publicKey })
    .instruction();
  return sendIxs(base.provider.connection, [ix], authority);
}

export async function joinTable(
  base: Program<HiddenHand>, payer: Keypair, player: Keypair, table: PublicKey, commitment: Uint8Array,
): Promise<string> {
  const ix = await base.methods
    .joinTable(Array.from(commitment))
    .accountsPartial({ table, hand: handPda(table, player.publicKey), vault: vaultPda(table), player: player.publicKey })
    .instruction();
  const extra = payer.publicKey.equals(player.publicKey) ? [] : [player];
  return sendIxs(base.provider.connection, [ix], payer, extra);
}

/** Base layer: delegate the table, then each hand, one tx each. Returns the base-layer signatures (proof). */
export async function delegatePdas(
  base: Program<HiddenHand>, payer: Keypair, roundId: BN, table: PublicKey, owners: PublicKey[], validator: PublicKey,
): Promise<string[]> {
  const conn = base.provider.connection;
  const sigs: string[] = [];
  const tIx = await base.methods.delegatePda({ table: { roundId } }).accountsPartial({ payer: payer.publicKey, validator, pda: table }).instruction();
  sigs.push(await sendIxs(conn, [tIx], payer));
  for (const owner of owners) {
    const ix = await base.methods
      .delegatePda({ hand: { table, owner } })
      .accountsPartial({ payer: payer.publicKey, validator, pda: handPda(table, owner) })
      .instruction();
    sigs.push(await sendIxs(conn, [ix], payer));
  }
  return sigs;
}

export async function requestDeal(er: Program<HiddenHand>, payer: Keypair, table: PublicKey, clientSeed: number): Promise<string> {
  const ix = await er.methods
    .requestDeal(clientSeed)
    .accountsPartial({ payer: payer.publicKey, table, oracleQueue: DEFAULT_EPHEMERAL_QUEUE })
    .instruction();
  return sendIxs(er.provider.connection, [ix], payer);
}
```

```ts
import { MAGIC_PROGRAM_ID, PERMISSION_PROGRAM_ID, permissionPdaFromAccount, waitUntilPermissionActive } from "@magicblock-labs/ephemeral-rollups-sdk";
import type { Connection, Keypair, PublicKey } from "@solana/web3.js";
import { EPHEMERAL_VAULT_ID, TEE_RPC, TEE_VALIDATOR } from "../config";
import { teeConnection } from "../conn";
import { handPda } from "../pda";
import { makeProgram } from "../program";
import { fetchTable, waitUntilOnEr } from "../round-base";
import { sendIxs } from "../tx";
import type { DealAdapter, DealContext } from "./types";

export class TeeDealAdapter implements DealAdapter {
  readonly mode = "tee" as const;
  readonly erRpc = TEE_RPC;
  readonly validator: PublicKey = TEE_VALIDATOR;
  private readonly conns = new Map<string, { connection: Connection; expiresAtMs: number }>();

  commitmentFor(): Uint8Array {
    return new Uint8Array(32);
  }

  async erConnection(identity: Keypair): Promise<Connection> {
    const key = identity.publicKey.toBase58();
    const hit = this.conns.get(key);
    if (hit && hit.expiresAtMs - Date.now() > 60_000) return hit.connection;
    const t = await teeConnection(identity);
    const expiresAtMs = t.expiresAt > 1e12 ? t.expiresAt : t.expiresAt * 1000;
    this.conns.set(key, { connection: t.connection, expiresAtMs });
    return t.connection;
  }

  async afterDelegate(ctx: DealContext): Promise<string[]> {
    const er = await this.erConnection(ctx.payer);
    const program = makeProgram(er, ctx.payer);
    const hands = ctx.seatOwners.map((o) => handPda(ctx.table, o));
    await waitUntilOnEr(er, ctx.table, 30_000);
    for (const h of hands) await waitUntilOnEr(er, h, 30_000);
    const common = {
      payer: ctx.payer.publicKey,
      permissionProgram: PERMISSION_PROGRAM_ID,
      ephemeralVault: EPHEMERAL_VAULT_ID,
      magicProgram: MAGIC_PROGRAM_ID,
    };
    const ixs = [
      await program.methods
        .createTablePermission()
        .accountsPartial({ table: ctx.table, permission: permissionPdaFromAccount(ctx.table), ...common })
        .instruction(),
    ];
    for (let i = 0; i < hands.length; i++) {
      ixs.push(
        await program.methods
          .createHandPermission(ctx.seatOwners[i])
          .accountsPartial({ table: ctx.table, hand: hands[i], permission: permissionPdaFromAccount(hands[i]), ...common })
          .instruction(),
      );
    }
    const sig = await sendIxs(er, ixs, ctx.payer);
    // returns false on timeout (never throws); a public permission (the table) has no members and never reports active
    for (const account of hands) {
      const ok = await waitUntilPermissionActive(TEE_RPC, account, 30_000);
      if (!ok) throw new Error(`permission not active: ${account.toBase58()}`);
    }
    return [sig];
  }

  async readOwnHand(ctx: DealContext, owner: Keypair): Promise<[number, number] | null> {
    const program = makeProgram(await this.erConnection(owner), owner);
    const hand = await program.account.hand.fetchNullable(handPda(ctx.table, owner.publicKey));
    if (!hand || !hand.dealt) return null;
    return [hand.cards[0], hand.cards[1]];
  }

  async reveal(ctx: DealContext): Promise<string[]> {
    const program = makeProgram(await this.erConnection(ctx.payer), ctx.payer);
    const view = await fetchTable(program, ctx.table);
    if (!view || view.phase !== "trading") return [];
    const h = view.seats.map((s) => s.hand);
    const ix = await program.methods
      .revealHands()
      .accountsPartial({
        table: ctx.table,
        hand0: h[0], hand1: h[1], hand2: h[2], hand3: h[3],
        permission0: permissionPdaFromAccount(h[0]),
        permission1: permissionPdaFromAccount(h[1]),
        permission2: permissionPdaFromAccount(h[2]),
        permission3: permissionPdaFromAccount(h[3]),
        payer: ctx.payer.publicKey,
        permissionProgram: PERMISSION_PROGRAM_ID,
        ephemeralVault: EPHEMERAL_VAULT_ID,
        magicProgram: MAGIC_PROGRAM_ID,
      })
      .instruction();
    return [await sendIxs(program.provider.connection, [ix], ctx.payer)];
  }
}
```

```ts
export type CheckId = "a" | "b" | "c" | "f";
export interface CheckResult { id: CheckId; pass: boolean; detail: string }

const ORDER: CheckId[] = ["a", "b", "c", "f"];
const TITLES: Record<CheckId, string> = {
  a: "TEE auth-token read of a permissioned account (owner reads, stranger cannot)",
  b: "VRF request + callback served inside the TEE ER (DEFAULT_EPHEMERAL_QUEUE)",
  c: "reveal of private state at settlement (reveal_hands flips every hand public)",
  f: "informational: VRF request + callback on devnet-as (the fallback path)",

export function formatSpikeResult(at: Date, programId: string, results: CheckResult[]): string {
  const byId = new Map(results.map((r) => [r.id, r]));
  const lines = ["# Spike result", "", `- run: ${at.toISOString()}`, `- program: ${programId}`, ""];
  for (const id of ORDER) {
    const r = byId.get(id) ?? { id, pass: false, detail: "not run" };
    const detail = r.detail.replace(/\s+/g, " ").trim().slice(0, 400);
    lines.push(`(${id}) ${r.pass ? "PASS" : "FAIL"} — ${TITLES[id]}: ${detail}`);
  }
  const eligible = ["a", "b", "c"].every((id) => byId.get(id as CheckId)?.pass === true);
  lines.push("", `TEE path eligible: ${eligible ? "yes — keep manifest.json" : "no — switch to manifest-fallback.json"}`, "");
  return lines.join("\n");
}
```

```ts
// Hidden Hand engine (TS port of program/engine). Integer cents. Pinned by program/engine-vectors.json.
export const START_PRICE_C = 6400;
export const STEP_C = 100;
export const MAX_LOT = 5;
export const POS_CAP = 15;
export const MIN_PRICE_C = 800;
export const MAX_PRICE_C = 12000;
export const SEATS = 4;
export const CARDS_PER_SEAT = 2;
export const HAND_MEAN = 16; // two cards, each uniform 1..15

export type Side = "buy" | "sell";
export type TradeError = "LotSize" | "PositionCap" | "PriceBound";
export interface SeatBook { cashC: number; pos: number }
export interface Fill { priceAfterC: number; cashDeltaC: number; posDelta: number }

export function quote(priceC: number, pos: number, side: Side, qty: number): Fill | TradeError {
  if (!Number.isInteger(qty) || qty < 1 || qty > MAX_LOT) return "LotSize";
  const impact = (STEP_C * qty * qty) / 2;
  if (side === "buy") {
    if (pos + qty > POS_CAP) return "PositionCap";
    const priceAfterC = priceC + STEP_C * qty;
    if (priceAfterC > MAX_PRICE_C) return "PriceBound";
    return { priceAfterC, cashDeltaC: -(qty * priceC + impact), posDelta: qty };
  }
  if (pos - qty < -POS_CAP) return "PositionCap";
  const priceAfterC = priceC - STEP_C * qty;
  if (priceAfterC < MIN_PRICE_C) return "PriceBound";
  return { priceAfterC, cashDeltaC: qty * priceC - impact, posDelta: -qty };
}

export function cardsFromRandomness(bytes: Uint8Array): number[] {
  const cards: number[] = [];
  for (let i = 0; i < SEATS * CARDS_PER_SEAT; i++) cards.push(1 + (bytes[i] % 15));
  return cards;
}

export function hiddenSum(cards: number[]): number {
  return cards.reduce((a, b) => a + b, 0);
}

export function pnlC(seat: SeatBook, sum: number): number {
  return seat.cashC + seat.pos * sum * 100;
}

export function winner(seats: SeatBook[], sum: number): number {
  let best = 0;
  for (let i = 1; i < seats.length; i++) if (pnlC(seats[i], sum) > pnlC(seats[best], sum)) best = i;
  return best;
}

/** What a sealed hand knows, in contracts: one contract moves the price one point. */
export function handEdge(cards: [number, number]): number {
  return cards[0] + cards[1] - HAND_MEAN;
}
```

```ts
    .settleRound()
```

```ts
import { HAND_MEAN, POS_CAP, type Side } from "@hidden-hand/client/engine";

export const DEMO_SEED = 20261127;

/** Seeded PRNG (mulberry32): the same seed gives the same sequence everywhere. */
export function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

export interface BotPersona {
  intervalMinMs: number;
  intervalMaxMs: number;
  rampMs: number;
  wobbleProb: number;
  maxLot: number;
}
export interface BotMemory { wobble: number }
export interface Order { side: Side; qty: number }

/** Target position = the hand's edge in contracts (1 contract = 1 point), phased in over rampMs, plus a -1/0/+1 wobble. */
export function botTarget(ownSum: number, tMs: number, persona: BotPersona, mem: BotMemory): number {
  const edge = ownSum - HAND_MEAN;
  const ramp = Math.min(1, tMs / persona.rampMs);
  const raw = Math.round(edge * ramp) + mem.wobble;
  return Math.max(-POS_CAP, Math.min(POS_CAP, raw));
}

export function decide(ownSum: number, pos: number, tMs: number, persona: BotPersona, mem: BotMemory, rng: () => number): Order | null {
  if (rng() < persona.wobbleProb) mem.wobble = Math.floor(rng() * 3) - 1;
  const target = botTarget(ownSum, tMs, persona, mem);
  const diff = target - pos;
  if (diff === 0) return null;
  const lot = 1 + Math.floor(rng() * persona.maxLot);
  return { side: diff > 0 ? "buy" : "sell", qty: Math.min(Math.abs(diff), lot) };
}

export function personaFor(seat: number, rng: () => number): BotPersona {
  return {
    intervalMinMs: 700 + seat * 100,
    intervalMaxMs: 1900 + seat * 150,
    rampMs: 15_000 + Math.floor(rng() * 30_000),
    wobbleProb: 0.25,
    maxLot: 3,
  };
}
```

```ts
import { CARDS_PER_SEAT, SEATS, START_PRICE_C, hiddenSum, quote, type SeatBook } from "@hidden-hand/client/engine";
import { type BotMemory, type BotPersona, type Order, decide, mulberry32, personaFor } from "./policy";

export type FourthSeat = "bot" | "slow-follower" | "random" | "idle";
export interface RoundOutcome { sum: number; closeC: number; trades: number }
export interface Summary { deals: number; within4: number; medianAbsErr: number; p90AbsErr: number; medianDoNothing: number; medianTrades: number }

interface Actor { seat: number; own: number; persona: BotPersona; mem: BotMemory; nextMs: number; kind: FourthSeat }

export function simulateRound(seed: number, roundMs: number, fourth: FourthSeat): RoundOutcome {
  const rng = mulberry32(seed);
  const cards: number[] = [];
  for (let i = 0; i < SEATS * CARDS_PER_SEAT; i++) cards.push(1 + Math.floor(rng() * 15));
  const books: SeatBook[] = Array.from({ length: SEATS }, () => ({ cashC: 0, pos: 0 }));
  let priceC = START_PRICE_C;
  let trades = 0;
  const actors: Actor[] = [];
  for (let seat = 0; seat < SEATS; seat++) {
    const kind: FourthSeat = seat === SEATS - 1 ? fourth : "bot";
    const persona = personaFor(seat, rng);
    if (kind === "slow-follower") {
      persona.intervalMinMs = 2000;
      persona.intervalMaxMs = 5000;
      persona.rampMs = 30_000 + Math.floor(rng() * 30_000);
    }
    const own = cards[seat * 2] + cards[seat * 2 + 1];
    actors.push({ seat, own, persona, mem: { wobble: 0 }, nextMs: Math.floor(rng() * persona.intervalMaxMs), kind });
  }
  for (;;) {
    let a = actors[0];
    for (const x of actors) if (x.nextMs < a.nextMs) a = x;
    if (a.nextMs >= roundMs) break;
    const t = a.nextMs;
    let order: Order | null = null;
    if (a.kind === "bot" || a.kind === "slow-follower") order = decide(a.own, books[a.seat].pos, t, a.persona, a.mem, rng);
    else if (a.kind === "random") order = { side: rng() < 0.5 ? "buy" : "sell", qty: 1 + Math.floor(rng() * 3) };
    if (order) {
      const fill = quote(priceC, books[a.seat].pos, order.side, order.qty);
      if (typeof fill !== "string") {
        priceC = fill.priceAfterC;
        books[a.seat].cashC += fill.cashDeltaC;
        books[a.seat].pos += fill.posDelta;
        trades++;
      }
    }
    const gap = a.kind === "random"
      ? 3000 + rng() * 5000
      : a.persona.intervalMinMs + rng() * (a.persona.intervalMaxMs - a.persona.intervalMinMs);
    a.nextMs = t + Math.floor(gap);
  }
  return { sum: hiddenSum(cards), closeC: priceC, trades };
}

export function summarize(firstSeed: number, deals: number, roundMs: number, fourth: FourthSeat): Summary {
  const errs: number[] = [];
  const idle: number[] = [];
  const tr: number[] = [];
  let within = 0;
  for (let i = 0; i < deals; i++) {
    const r = simulateRound(firstSeed + i, roundMs, fourth);
    const e = Math.abs(r.closeC / 100 - r.sum);
    errs.push(e);
    idle.push(Math.abs(64 - r.sum));
    tr.push(r.trades);
    if (e <= 4) within++;
  }
  const sorted = (v: number[]) => [...v].sort((x, y) => x - y);
  const med = (v: number[]) => sorted(v)[Math.floor(v.length / 2)];
  const p90 = (v: number[]) => sorted(v)[Math.floor(v.length * 0.9)];
  return { deals, within4: within / deals, medianAbsErr: med(errs), p90AbsErr: p90(errs), medianDoNothing: med(idle), medianTrades: med(tr) };
}
```

The runner is run **only by human gates** (it signs devnet transactions with the burner). Executors write and unit-test it; never run `demo-round` yourself. The proof must carry **base-layer** signatures (open, joins, delegations, the validator's commit/undelegate landings, claim) as well as ER ones, because the portal may scan only the base layer; never write a TEE tokenized URL into the proof. Seat 4 is a bot unless `--seat4 <pubkey>` names a browser seat that joins within 20 s; the README states that a random or idle human breaks the reveal.

```ts
export interface ProofData {
  generatedAt: string; mode: "tee" | "commit-reveal"; programId: string; table: string; roundId: string; validator: string; erRpc: string;
  base: { openTable: string; joins: string[]; delegations: string[]; commitUndelegate: string[]; claimPot: string };
  er: { setup: string[]; requestDeal: string; trades: string[]; reveal: string[]; settle: string };
  result: { hiddenSum: number; closeC: number; winnerSeat: number; winner: string; pnlC: number[]; potLamports: number; cards: number[] };
}

const tx = (s: string) => `https://explorer.solana.com/tx/${s}?cluster=devnet`;
const addr = (a: string) => `https://explorer.solana.com/address/${a}?cluster=devnet`;
const SIG = "[1-9A-HJ-NP-Za-km-z]{64,90}";

export function formatProof(d: ProofData): string {
  const row = (step: string, s: string) => `| ${step} | \`${s}\` | [explorer](${tx(s)}) |`;
  const L: string[] = [
    "# Devnet round proof", "",
    `- generated: ${d.generatedAt}`,
    `- deal mode: ${d.mode}`,
    `- program: ${d.programId} ([explorer](${addr(d.programId)}))`,
    `- table PDA: ${d.table}`,
    `- round id: ${d.roundId}`,
    `- ER: ${d.erRpc} (validator ${d.validator})`, "",
    "## Base layer (Solana devnet)", "",
    "| step | signature | explorer |", "|---|---|---|",
    row("open_table", d.base.openTable),
    ...d.base.joins.map((s, i) => row(`join_table seat ${i}`, s)),
    ...d.base.delegations.map((s, i) => row(i === 0 ? "delegate_pda table" : `delegate_pda hand ${i - 1}`, s)),
    ...d.base.commitUndelegate.map((s) => row("commit/undelegate (validator)", s)),
    row("claim_pot", d.base.claimPot), "",
    "## Ephemeral Rollup", "",
    "ER signatures are read through the ER RPC above (the TEE needs an auth token; none is stored here).", "",
    "| step | signature |", "|---|---|",
    ...d.er.setup.map((s) => `| permissions / setup | \`${s}\` |`),
    `| request_deal (VRF) | \`${d.er.requestDeal}\` |`,
    `| trades | ${d.er.trades.length} trades; first \`${d.er.trades[0] ?? "none"}\` |`,
    ...d.er.reveal.map((s) => `| reveal | \`${s}\` |`),
    `| settle_round | \`${d.er.settle}\` |`, "",
    "## Result", "",
    `- hidden sum: ${d.result.hiddenSum}`,
    `- close: ${(d.result.closeC / 100).toFixed(2)} (|close − sum| = ${Math.abs(d.result.closeC / 100 - d.result.hiddenSum).toFixed(2)})`,
    `- cards: ${d.result.cards.join(", ")}`,
    `- winner: seat ${d.result.winnerSeat} (${d.result.winner}), pot ${(d.result.potLamports / 1e9).toFixed(3)} SOL`,
    `- P&L (points): ${d.result.pnlC.map((p) => (p / 100).toFixed(2)).join(", ")}`, "",
    `Explorer link for the submission form (a base-layer delegation of our program): ${tx(d.base.delegations[0] ?? "")}`, "",
  ];
  return L.join("\n");
}

export function checkProof(text: string): { ok: boolean; missing: string[] } {
  const need: [string, RegExp][] = [
    ["program id", /^- program: [1-9A-HJ-NP-Za-km-z]{32,44} /m],
    ["base delegate_pda table signature", new RegExp(`^\\| delegate_pda table \\| \`${SIG}\``, "m")],
    ["base commit/undelegate signature", new RegExp(`^\\| commit/undelegate \\(validator\\) \\| \`${SIG}\``, "m")],
    ["base claim_pot signature", new RegExp(`^\\| claim_pot \\| \`${SIG}\``, "m")],
    ["request_deal signature", new RegExp(`^\\| request_deal \\(VRF\\) \\| \`${SIG}\``, "m")],
    ["hidden sum", /^- hidden sum: \d+$/m],
    ["winner", /^- winner: seat [0-3] /m],
  ];
  const missing = need.filter(([, re]) => !re.test(text)).map(([name]) => name);
  return { ok: missing.length === 0, missing };
}
```

```ts
import { writeFileSync, mkdirSync } from "node:fs";
import { resolve } from "node:path";
import BN from "bn.js";
import {
  DEFAULT_DEAL_MODE, type DealContext, type DealMode, ENTRY_LAMPORTS, PROGRAM_ID, type TableView, baseConnection, claimPot, createDealAdapter,
  delegatePdas, deriveBotKeypair, fetchTable, findCommitSignatures, fundAccounts, joinTable, makeProgram, memorySecretStore,
  openTable, requestDeal, sendTrade, settleRound, tablePda, waitForOwner, waitForPhase,
} from "@hidden-hand/client";
import { REPO_ROOT, loadBurner } from "@hidden-hand/client/node";
import { type Keypair, LAMPORTS_PER_SOL, PublicKey } from "@solana/web3.js";
import { DEMO_SEED, type BotMemory, decide, mulberry32, personaFor } from "./policy";
import { type ProofData, formatProof } from "./proof";

export interface RunOptions { demo: boolean; proof: boolean; seat4: PublicKey | null; durationSecs: number; mode: DealMode; joinWindowSecs: number }
const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
const log = (m: string) => console.log(`[demo-round ${new Date().toISOString()}] ${m}`);

export async function runDemoRound(o: RunOptions): Promise<ProofData> {
  const burner = loadBurner();
  const base = baseConnection();
  const baseProgram = makeProgram(base, burner);
  const adapter = createDealAdapter(o.mode, memorySecretStore());
  const bots = [0, 1, 2, 3].map((i) => deriveBotKeypair(burner, i));
  const rng = mulberry32(o.demo ? DEMO_SEED : (Date.now() >>> 0));
  const roundId = new BN(Date.now());
  const table = tablePda(roundId);

  const openSig = await openTable(baseProgram, burner, roundId, o.mode, o.durationSecs, ENTRY_LAMPORTS);
  log(`table ${table.toBase58()} open; watch it at /app/?round=${roundId.toString()}`);
  const fundTargets = bots.map((b) => b.publicKey).concat(o.seat4 ? [o.seat4] : []);
  await fundAccounts(base, burner, fundTargets, 0.03 * LAMPORTS_PER_SOL);

  const joins: string[] = [];
  for (const b of bots.slice(0, 3)) joins.push(await joinTable(baseProgram, burner, b, table, adapter.commitmentFor(table, b.publicKey)));
  let humanJoined = false;
  if (o.seat4) {
    log(`waiting ${o.joinWindowSecs}s for the browser seat ${o.seat4.toBase58()}`);
    const deadline = Date.now() + o.joinWindowSecs * 1000;
    while (Date.now() < deadline) {
      const v = await fetchTable(baseProgram, table);
      if (v && v.seatCount === 4) { humanJoined = true; break; }
      await sleep(1000);
    }
  }
  if (!humanJoined) joins.push(await joinTable(baseProgram, burner, bots[3], table, adapter.commitmentFor(table, bots[3].publicKey)));

  const seated = await fetchTable(baseProgram, table);
  if (!seated || seated.seatCount !== 4) throw new Error("table did not fill four seats");
  const owners = seated.seats.map((s) => s.owner);
  const ctx: DealContext = { roundId, table, seatOwners: owners, payer: burner };
  const delegations = await delegatePdas(baseProgram, burner, roundId, table, owners, adapter.validator);
  const setup = await adapter.afterDelegate(ctx);

  const erBurner = makeProgram(await adapter.erConnection(burner), burner);
  const dealSig = await requestDeal(erBurner, burner, table, Math.floor(rng() * 256));
  let view: TableView = await waitForPhase(erBurner, table, "trading", 45_000);
  log(`dealt; trading until ${new Date(view.endTs * 1000).toISOString()}`);

  const botSeats = owners.map((ow, seat) => ({ seat, key: bots.find((b) => b.publicKey.equals(ow)) ?? null })).filter((x): x is { seat: number; key: Keypair } => x.key !== null);
  const ers = await Promise.all(botSeats.map(async (b) => makeProgram(await adapter.erConnection(b.key), b.key)));
  const own = await Promise.all(botSeats.map(async (b) => {
    const h = await adapter.readOwnHand(ctx, b.key);
    if (!h) throw new Error(`seat ${b.seat} cannot read its own hand`);
    return h[0] + h[1];
  }));
  const personas = botSeats.map((b) => personaFor(b.seat, rng));
  const mems: BotMemory[] = botSeats.map(() => ({ wobble: 0 }));
  const startMs = view.startTs * 1000;
  const endMs = view.endTs * 1000;
  const next = personas.map((p) => Date.now() + Math.floor(rng() * p.intervalMaxMs));
  const trades: string[] = [];
  let lastRefresh = 0;
  while (Date.now() < endMs - 1500) {
    let i = 0;
    for (let k = 1; k < next.length; k++) if (next[k] < next[i]) i = k;
    const wait = next[i] - Date.now();
    if (wait > 0) await sleep(wait);
    if (Date.now() >= endMs - 1500) break;
    if (Date.now() - lastRefresh > 800) {
      view = (await fetchTable(erBurner, table).catch(() => null)) ?? view;
      lastRefresh = Date.now();
    }
    const seat = botSeats[i].seat;
    const order = decide(own[i], view.seats[seat].pos, Date.now() - startMs, personas[i], mems[i], rng);
    if (order) {
      try {
        trades.push(await sendTrade(ers[i], botSeats[i].key, table, order.side, order.qty));
        view.seats[seat].pos += order.side === "buy" ? order.qty : -order.qty;
      } catch (e) {
        log(`seat ${seat} trade skipped: ${e instanceof Error ? e.message.slice(0, 120) : String(e)}`);
      }
    }
    const p = personas[i];
    next[i] = Date.now() + Math.floor(p.intervalMinMs + rng() * (p.intervalMaxMs - p.intervalMinMs));
  }

  const bellAt = endMs + 1000;
  if (Date.now() < bellAt) await sleep(bellAt - Date.now());
  log("bell");
  const reveal = await adapter.reveal(ctx, botSeats.map((b) => b.key));
  view = await waitForPhase(erBurner, table, "revealed", 60_000);
  const settleSig = await settleRound(erBurner, burner, view);
  await waitForOwner(base, table, PROGRAM_ID, 120_000);
  const settled = await waitForPhase(baseProgram, table, "settled", 30_000);
  const claimSig = await claimPot(baseProgram, burner, settled);
  const known = [openSig, ...joins, ...delegations, claimSig];
  const commitUndelegate = await findCommitSignatures(base, table, known);
  log(`hidden sum ${settled.hiddenSum}, close ${(settled.priceC / 100).toFixed(2)}, winner seat ${settled.winner}`);

  const data: ProofData = {
    generatedAt: new Date().toISOString(), mode: o.mode, programId: PROGRAM_ID.toBase58(), table: table.toBase58(),
    roundId: roundId.toString(), validator: adapter.validator.toBase58(), erRpc: adapter.erRpc,
    base: { openTable: openSig, joins, delegations, commitUndelegate, claimPot: claimSig },
    er: { setup, requestDeal: dealSig, trades, reveal, settle: settleSig },
    result: {
      hiddenSum: settled.hiddenSum, closeC: settled.priceC, winnerSeat: settled.winner ?? 0,
      winner: settled.seats[settled.winner ?? 0].owner.toBase58(), pnlC: settled.pnlC,
      potLamports: ENTRY_LAMPORTS * 4, cards: settled.cards,
    },
  };
  if (o.proof) {
    mkdirSync(resolve(REPO_ROOT, "docs/proof"), { recursive: true });
    writeFileSync(resolve(REPO_ROOT, "docs/proof/devnet-round.md"), formatProof(data), "utf8");
    log("wrote docs/proof/devnet-round.md");
  }
  return data;
}

export function parseArgs(argv: string[]): RunOptions {
  const has = (f: string) => argv.includes(f);
  const val = (f: string) => { const i = argv.indexOf(f); return i >= 0 ? argv[i + 1] : undefined; };
  const demo = has("--demo");
  const seat4 = val("--seat4");
  const mode = (val("--mode") ?? DEFAULT_DEAL_MODE) as DealMode;
  return {
    demo,
    proof: has("--proof"),
    seat4: seat4 ? new PublicKey(seat4) : null,
    durationSecs: Number(val("--duration") ?? (demo ? 75 : 90)),
    mode,
    joinWindowSecs: 20,
  };
}
```
