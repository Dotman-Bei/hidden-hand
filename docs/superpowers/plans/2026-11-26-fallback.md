# Commit-Reveal Fallback Implementation Plan (delta)

**Goal:** If the Day-1 spike fails, keep sealed hands, VRF and the ER without the TEE: each seat commits `sha256("hidden-hand-commit" ‖ secret)` at join, the VRF deal stores a public seed, each hand is `sha256("hidden-hand-cards" ‖ seed ‖ secret ‖ seat)` known only to its owner until the owner reveals the secret at the bell.
**Architecture:** A purely additive delta on the same program and client: `program:20` adds `callback_deal_cr` and `reveal_secret` (the `mode`/`commitment`/`seed`/`revealed` fields already exist since Task 1; settlement and claim are unchanged), `client:21` adds the `CrDealAdapter` and flips the default deal mode. The fallback ER is the public `devnet-as` (validator `MAS1Dt9qreoRMQ14YQuhg8UTZMMzDdKhmkZMECCzk57`), VRF still on `DEFAULT_EPHEMERAL_QUEUE`.
**Tech Stack:** as the program and clients plans, plus `solana-sha256-hasher = { version = "3.1.0", features = ["sha2"] }` (syscall on-chain, `sha2` on the host for tests) and `@noble/hashes` ^1.8.0 (already a client dependency).
**Source spec:** docs/superpowers/specs/2026-11-26-hidden-hand-design.md

```rust
/// Commit-reveal fallback: the 32-byte commitment a seat sends at join.
pub fn commitment_of(secret: &[u8; 32]) -> [u8; 32] {
    solana_sha256_hasher::hashv(&[b"hidden-hand-commit".as_ref(), secret.as_ref()]).to_bytes()
}

/// Commit-reveal fallback: a seat's two cards from the public VRF seed and its private secret.
pub fn cards_for_seat(seed: &[u8; 32], secret: &[u8; 32], seat: u8) -> [u8; 2] {
    let h = solana_sha256_hasher::hashv(&[b"hidden-hand-cards".as_ref(), seed.as_ref(), secret.as_ref(), &[seat]]).to_bytes();
    [1 + h[0] % CARD_MAX, 1 + h[1] % CARD_MAX]
}
```

```rust
    /// ER (VRF callback, commit-reveal mode). Store the public seed; a hand needs its owner's secret.
    pub fn callback_deal_cr(ctx: Context<CallbackDealCr>, randomness: [u8; 32]) -> Result<()> {
        let table = &mut ctx.accounts.table;
        require!(table.phase == Phase::Dealing, HhError::WrongPhase);
        require!(table.mode == DealMode::CommitReveal, HhError::ModeNotSupported);
        table.seed = randomness;
        let now = Clock::get()?.unix_timestamp;
        table.start_ts = now;
        table.end_ts = now + table.duration_secs as i64;
        table.phase = Phase::Trading;
        msg!("hh:dealt-cr round={} end_ts={}", table.round_id, table.end_ts);
        Ok(())
    }

    /// ER (commit-reveal). After the bell each seat reveals its secret; the program checks it
    /// against the join-time commitment and derives the cards. The last reveal sets Revealed.
    pub fn reveal_secret(ctx: Context<RevealSecret>, secret: [u8; 32]) -> Result<()> {
        let now = Clock::get()?.unix_timestamp;
        let owner = ctx.accounts.owner.key();
        let table = &mut ctx.accounts.table;
        require!(table.mode == DealMode::CommitReveal, HhError::ModeNotSupported);
        require!(table.phase == Phase::Trading, HhError::WrongPhase);
        require!(now >= table.end_ts, HhError::BellNotRung);
        let seat = table
            .seats
            .iter()
            .position(|s| s.owner == owner)
            .ok_or(HhError::NotSeated)?;
        require!(!table.seats[seat].revealed, HhError::AlreadyRevealed);
        require!(commitment_of(&secret) == table.seats[seat].commitment, HhError::BadCommitment);
        let cards = cards_for_seat(&table.seed, &secret, seat as u8);
        table.cards[2 * seat] = cards[0];
        table.cards[2 * seat + 1] = cards[1];
        table.seats[seat].revealed = true;
        if table.seats.iter().all(|s| s.revealed) {
            table.hidden_sum = hidden_sum(&table.cards);
            table.phase = Phase::Revealed;
        }
        let hand = &mut ctx.accounts.hand;
        hand.cards = cards;
        hand.dealt = true;
        msg!("hh:reveal-secret seat={} cards={:?}", seat, cards);
        Ok(())
    }
```

```ts
import { sha256 } from "@noble/hashes/sha256";
import type { Connection, Keypair, PublicKey } from "@solana/web3.js";
import { AS_RPC, AS_VALIDATOR } from "../config";
import { publicErConnection } from "../conn";
import { handPda } from "../pda";
import { makeProgram } from "../program";
import { fetchTable, waitUntilOnEr } from "../round-base";
import { sendIxs } from "../tx";
import type { DealAdapter, DealContext, SecretStore } from "./types";

const enc = new TextEncoder();
const concat = (...parts: Uint8Array[]) => {
  const out = new Uint8Array(parts.reduce((n, p) => n + p.length, 0));
  let o = 0;
  for (const p of parts) { out.set(p, o); o += p.length; }
  return out;
};

export function commitmentOf(secret: Uint8Array): Uint8Array {
  return sha256(concat(enc.encode("hidden-hand-commit"), secret));
}

export function cardsForSeat(seed: Uint8Array, secret: Uint8Array, seat: number): [number, number] {
  const h = sha256(concat(enc.encode("hidden-hand-cards"), seed, secret, Uint8Array.of(seat)));
  return [1 + (h[0] % 15), 1 + (h[1] % 15)];
}

export class CrDealAdapter implements DealAdapter {
  readonly mode = "commit-reveal" as const;
  readonly erRpc = AS_RPC;
  readonly validator: PublicKey = AS_VALIDATOR;
  private readonly er: Connection = publicErConnection(AS_RPC);
  constructor(private readonly store: SecretStore) {}

  private key(table: PublicKey, owner: PublicKey): string {
    return `hh-secret:${table.toBase58()}:${owner.toBase58()}`;
  }

  commitmentFor(table: PublicKey, owner: PublicKey): Uint8Array {
    const k = this.key(table, owner);
    let secret = this.store.get(k);
    if (!secret) {
      secret = new Uint8Array(32);
      globalThis.crypto.getRandomValues(secret);
      this.store.set(k, secret);
    }
    return commitmentOf(secret);
  }

  async erConnection(): Promise<Connection> {
    return this.er;
  }

  async afterDelegate(ctx: DealContext): Promise<string[]> {
    await waitUntilOnEr(this.er, ctx.table, 30_000);
    for (const o of ctx.seatOwners) await waitUntilOnEr(this.er, handPda(ctx.table, o), 30_000);
    return [];
  }

  async readOwnHand(ctx: DealContext, owner: Keypair): Promise<[number, number] | null> {
    const secret = this.store.get(this.key(ctx.table, owner.publicKey));
    if (!secret) return null;
    const view = await fetchTable(makeProgram(this.er, owner), ctx.table);
    if (!view || view.phase === "open" || view.phase === "dealing") return null;
    const raw = await makeProgram(this.er, owner).account.table.fetch(ctx.table);
    const seat = view.seats.findIndex((s) => s.owner.equals(owner.publicKey));
    if (seat < 0) return null;
    return cardsForSeat(Uint8Array.from(raw.seed), secret, seat);
  }

  async reveal(ctx: DealContext, seatKeys: Keypair[]): Promise<string[]> {
    const sigs: string[] = [];
    for (const kp of seatKeys) {
      const program = makeProgram(this.er, kp);
      const view = await fetchTable(program, ctx.table);
      if (!view || view.phase !== "trading") continue;
      const seat = view.seats.findIndex((s) => s.owner.equals(kp.publicKey));
      if (seat < 0 || view.seats[seat].revealed) continue;
      const secret = this.store.get(this.key(ctx.table, kp.publicKey));
      if (!secret) throw new Error(`no stored secret for seat ${seat}`);
      const ix = await program.methods
        .revealSecret(Array.from(secret))
        .accountsPartial({ table: ctx.table, hand: handPda(ctx.table, kp.publicKey), owner: kp.publicKey })
        .instruction();
      sigs.push(await sendIxs(this.er, [ix], kp));
    }
    return sigs;
  }
}
```
